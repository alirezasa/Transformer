self.W_q = nn.Linear(d_model, d_model)  # also W_k, W_v, W_o

def _split_heads(self, x):   # (B,T,d) -> (B,h,T,d_k)
  B, T, _ = x.shape
  return x.view(B, T, self.h, self.d_k).transpose(1, 2)

def _merge_heads(self, x):   # (B,h,T,d_k) -> (B,T,d)
  return x.transpose(1, 2).contiguous().view(B, T, -1)

def forward(self, x, key_mask=None):
  q = self._split_heads(self.W_q(x))
  k = self._split_heads(self.W_k(x))
  v = self._split_heads(self.W_v(x))
  out, attn = self.attention(q, k, v, key_mask)
  return self.W_o(self._merge_heads(out)), attn

توجه چندسر به جای یک توجه بزرگ، چهار توجه کوچک‌تر را موازی اجرا می‌کند. پیاده‌سازی من چهار لایه‌ی خطی دارد: W_q و W_k و W_v که ورودی را به query و key و value نگاشت می‌کنند و W_o که در آخر خروجی سرها را ترکیب می‌کند. نکته‌ی اصلی تابع split_heads است: خروجی لایه‌ی خطی ابعاد B در T در ۱۲۸ دارد؛ با view آن را به B در T در ۴ در ۳۲ تبدیل می‌کنم، یعنی ۱۲۸ بعد را به چهار تکه‌ی ۳۲ تایی می‌شکنم، و بعد با transpose بعد سر را جلو می‌آورم تا بشود B در ۴ در T در ۳۲. حالا همان کلاس توجه اسلاید قبل روی هر چهار سر به‌صورت موازی و بدون حلقه اجرا می‌شود، چون ضرب ماتریسی روی دو بعد آخر عمل می‌کند. merge_heads برعکس همین کار است: transpose، بعد contiguous چون بعد از transpose حافظه پیوسته نیست و view خطا می‌دهد، و بعد view به B در T در ۱۲۸. در آخر W_o اعمال می‌شود. چون Q و K و V هر سه از خود x ساخته می‌شوند، این Self-Attention است. فایده‌ی چند سر این است که هر سر می‌تواند روی نوع متفاوتی از رابطه تمرکز کند؛ در heatmapها می‌بینیم که واقعاً همین اتفاق افتاده.
